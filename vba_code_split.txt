    gNo_Boats = Range("'Score Sheet'!A9").Value     ' Names("No_Boats").Value

    gSumRowMin = 3
    gSumRowMax = 90
    gSumSkipMin = gSumRowMin + 3
    gSumSkipMax = gSumRowMin + 2 + Range("'Score Sheet'!A9").Value
    ' 
If number of contestants is small and races are not divided into heats then Race 1 is included
    ' in tie-breaker data.  We'll test on C29 in Race Results because it holds the number of entries
    ' in Heat B.  The gHMS variable is used in Get_Data to determine first column of array to include
    ' in tie-breaker calculations
    gHMS = 1                                                 ' default is HMS
    
If Range("'Race Results'!C32").Value < 1 Then gHMS = 0   ' if no entries in B heat
    gSumColPoints = 5                                        ' column E
    gSumColTieNo = 7                                         ' column G on Summary sheet
    gSumColTieRef = 8                                        ' column H on Summary sheet
    gSumColSailNo = 9                                        ' column I on Summary sheet

    ' for BestScores we only display races with results [gNo_Races-gHMS],
    ' for Reverse order listing we'll display all races [gLast_Race to Race #1] including

Sub prepareSummarySheet()

    Application.Run "ClearSummary"
    Application.Run "CopyMainToSum"
    Application.CutCopyMode = False
    Application.Run "CopyPosToSum"
    Application.Run "SortSummary"
    
    ' look for ties ( '=' sign in column A) and invoke tie-breaker if we have at least 2 races
    
If Not TypeName(ActiveSheet) = "Summary" Then Sheets("Summary").Activate
    
If gNo_Races > 1 Then                                   ' gNo_Races set during ClearSummary
        
For iRow = gSumSkipMin To gSumSkipMax
            
If Range("A" & iRow).Value = "=" Then
                gStill_tied = False
                Application.Run "Number_the_ties"
                    Exit For
            
End If
        
Next iRow
    
End If

End Sub

Sub Number_the_ties()
    
Dim rRange As Range
    
Dim counter As Integer, tie_ref As Integer
    
Dim n As Integer, x As Integer, BestScores As Integer
    
Dim position As Integer, FirstR As Integer, LastR As Integer, FirstC As Integer, LastC As Integer, SortC As Integer
    
    
If gNo_Races < 2 Then Exit 
Sub      ' must have results from two or more races with heats
    
    ' to avoid redundant calculations, exit if we have not changed the Race results
    ' sheet since we last broke ties; exception: if ties still exist, perform tie
    ' break again so that user is advised of what needs to be done (toss for it)
    
    
If Not gChangeRaceResults Then Exit Sub
        
    BestScores = gNo_Races - gHMS           ' number of races used in tie breaking - includes discards
        
    'Macro to find boats with tied scores
    'Groups of boats having same score are numbered (gTie_Number)
    'Boats within the group are numbered (tie_ref)
    'Ties are then broken and boats re-ordered
    '
If tie break is not possible, message "Toss for it!" is displayed.

    gTColour = 12    'Red text for use in unbroken ties
        
    
If Not TypeName(ActiveSheet) = "Summary" Then Sheets("Summary").Activate
    Call Clear_Summary_Work_Area    'Clear workspace
    
    counter = 0
    gTie_Number = 0
    tie_ref = 0
    gMax_Tie_Ref = 0
    
    'Label header row for Tie Break Data
    
Set rRange = Range(Cells(gSumRowMin, gSumColBestL), Cells(gSumRowMin, gSumColRevR))
    rRange.Merge
    rRange.HorizontalAlignment = xlCenter
    rRange.Value = "TIE BREAK DATA"
    rRange.Font.Bold = True
    rRange.Font.Size = 8
    rRange.Interior.colorIndex = 15           ' gray
    
    
Set rRange = Range(Cells(gSumRowMin + 2, gSumColBestL), Cells(gSumRowMin + 2, gSumColBestR))
    rRange.Merge
    rRange.Interior.colorIndex = 13           ' purple
    
If gHMS = 1 Then
        rRange.Value = "Best Scores excl. R1"
    Else
        rRange.Value = "Best Scores"
    
End If
    rRange.Font.Bold = True
    rRange.Font.Size = 8
    rRange.Font.Color = QBColor(14)
    
    
Set rRange = Range(Cells(gSumRowMin + 2, gSumColRevL), Cells(gSumRowMin + 2, gSumColRevR))
    rRange.Merge
    rRange.Value = "Finishes in Reverse Order"
    rRange.Font.Color = QBColor(9)
    rRange.Font.Bold = True
    rRange.Font.Size = 8
    rRange.Interior.colorIndex = 6
        
    
Set rRange = Range(Cells(gSumRowMin + 2, gSumColTieNo), Cells(gSumRowMin + 3, gSumColTieNo))
    rRange.Merge
    rRange.WrapText = True
    rRange.Value = "Tie No."
    rRange.Font.Bold = True
    
    
Set rRange = Range(Cells(gSumRowMin + 2, gSumColTieRef), Cells(gSumRowMin + 3, gSumColTieRef))
    rRange.Merge
    rRange.WrapText = True
    rRange.Value = "Tie Ref."
    rRange.Font.Bold = True

    
Set rRange = Range(Cells(gSumRowMin + 2, gSumColSailNo), Cells(gSumRowMin + 3, gSumColSailNo))
    rRange.Merge
    rRange.WrapText = True
    rRange.Value = "Sail #"
    rRange.Font.Bold = True
   
    ' Write Best Score No.s to Summary sheet heading (these are NOT race numbers)
    
Set rRange = Cells(gSumRowMin + 3, gSumColBestL)
    rRange.Value = "Best"                         ' label first column 'Best'
    rRange.Font.Size = 8
    rRange.Font.Color = QBColor(14)
    rRange.Interior.colorIndex = 13
    
For n = 2 To BestScores                       ' start at 2
        
Set rRange = Cells(gSumRowMin + 3, gSumColBestL - 1 + n)
        rRange.Value = n
        rRange.Font.Color = QBColor(14)           ' yellow
        rRange.ColumnWidth = 4
        rRange.Font.Bold = True
        rRange.Interior.colorIndex = 13           ' purple
    
Next n
    
    'Write Race No.s in reverse order to Summary sheet heading - void races will show empty
    
For n = gLast_Race To 1 Step -1
        
Set rRange = Cells(gSumRowMin + 3, gSumColRevR + 1 - n)
        rRange.Value = n
        rRange.Font.Color = QBColor(9) ' blue
        rRange.ColumnWidth = 4
        rRange.Font.Bold = True
        rRange.Font.Size = 8
        rRange.Interior.colorIndex = 6 ' yellow
    
Next n
    
    ' set values of variables used in Tie_Breaker routine
    FirstC = 2                                      ' column B
    LastC = gSumColRevR                             ' Race #1 in list of reverse finishes
    SortC = gSumColBestL                            ' column J
    
    Do
        ' step past un-tied boats
        
Do 
While counter < gNo_Boats
            counter = counter + 1
            
If Cells(counter + gSumRowMin + 3, gSumColPoints).Value = Cells(counter + gSumRowMin + 4, gSumColPoints).Value Then '1st tied boat in group
                gTie_Number = gTie_Number + 1
                tie_ref = tie_ref + 1
                Cells(counter + gSumRowMin + 3, gSumColTieNo).Value = gTie_Number
                Cells(counter + gSumRowMin + 3, gSumColTieNo).HorizontalAlignment = xlCenter
                Cells(counter + gSumRowMin + 3, gSumColTieRef).Value = tie_ref
                Cells(counter + gSumRowMin + 3, gSumColTieRef).HorizontalAlignment = xlCenter
                Exit Do
            
End If
        Loop
         
        ' mark tied boats
        
Do 
While counter < gNo_Boats
            counter = counter + 1
            
If Not Cells(counter + gSumRowMin + 3, gSumColPoints).Value = Cells(counter + gSumRowMin + 4, gSumColPoints).Value Then 'un-tied boats
                tie_ref = tie_ref + 1
                Cells(counter + gSumRowMin + 3, gSumColTieNo).Value = gTie_Number
                Cells(counter + gSumRowMin + 3, gSumColTieNo).HorizontalAlignment = xlCenter
                Cells(counter + gSumRowMin + 3, gSumColTieRef).Value = tie_ref
                Cells(counter + gSumRowMin + 3, gSumColTieRef).HorizontalAlignment = xlCenter
                gMax_Tie_Ref = tie_ref
                tie_ref = 0
                
                ' Get the data to break ties
                Get_Data counter + 7 - gMax_Tie_Ref

                'Break Ties
                position = Cells(counter + 7 - gMax_Tie_Ref, 2).Value
                FirstR = counter + 7 - gMax_Tie_Ref
                LastR = counter + 6
                Break_Ties position, FirstR, LastR, FirstC, LastC, SortC
                Exit Do
            
End If
            tie_ref = tie_ref + 1
            Cells(counter + gSumRowMin + 3, gSumColTieNo).Value = gTie_Number
            Cells(counter + gSumRowMin + 3, gSumColTieNo).HorizontalAlignment = xlCenter
            Cells(counter + gSumRowMin + 3, gSumColTieRef).Value = tie_ref
            Cells(counter + gSumRowMin + 3, gSumColTieRef).HorizontalAlignment = xlCenter
        Loop
    
Loop Until counter = gNo_Boats
    
    
If gStill_tied = True Then          'added HCF 21/10/04
       x = MsgBox("Some ties cannot be broken - " & Chr$(13) & "
If event complete, toss for them and enter" & Chr$(13) & "order (1,2,3,..) in Man TB column on Score Sheet", vbOKOnly, "UNBROKEN TIES")
       gChangeRaceResults = True
    
End If
    gChangeRaceResults = False  ' reset to false

End Sub


Sub Get_Data(First As Integer)
    '
    
Dim n As Integer, m As Integer, p As Integer, offset As Integer, row As Integer, discard As Integer
    
Dim ResultColR1 As Integer, ResultColR2 As Integer, BestScores As Integer
    
Dim testValue As Variant
    
Dim remainder As Long
    
Dim rRange As Range
    
ReDim Race_results(gMax_Tie_Ref, (2 * gLast_Race - gRDiff) + 1) As Variant
    
    
If Not TypeName(ActiveSheet) = "Score Sheet" Then Sheets("Score Sheet").Activate
    
    ResultColR1 = 9                      ' column I
    ResultColR2 = ResultColR1 + gHMS     ' column J - first column to include for tie-break calculations
    
    BestScores = gNo_Races - gHMS        ' includes discards
           
    
For n = 1 To gMax_Tie_Ref
        Race_results(n, 1) = Worksheets("Summary").Cells(First - 1 + n, 4).Value    'sail no.
        ' find row on Score Sheet that holds this sail number
        row = (Application.WorksheetFunction.Match(Race_results(n, 1), Range("C10:C92"), 0)) + 9
                
        ' define range on Score Sheet from first race (depends on gHMS) through last race
        
Set rRange = Range(Cells(row, ResultColR2), Cells(row, ResultColR1 - 1 + gLast_Race))
        ' extract best order from this row of finishes -- first identify races to discard
        offset = 0
        discard = 0
        
For m = 1 To gNo_Races - gHMS                                  ' this range includes discards
            testValue = Application.WorksheetFunction.Large(rRange, m)
            Race_results(n, 1 + BestScores - offset) = Int(testValue * 10) / 10
            offset = offset + 1
        
Next m
        
        ' 
Next store race scores in reverse order - all races included except Race 1 in HMS mode
        ' strip off .000333 and .000334 by storing integer value of score
        ' this will be used to determine who beat who in last race - RRS A8.2
        p = 1
        
For m = 1 To gLast_Race
            
If Not (Cells(row, ResultColR1 - 1 + m) = "") Then          'STEPS OVER VOID SCORE
                Race_results(n, BestScores + gLast_Race + 2 - m) = Int(Cells(row, ResultColR1 - 1 + m) * 10) / 10
                p = p + 1
            
End If
        
Next m
    
Next n
        
    'write scores for tied boats to Summary sheet
    Sheets("Summary").Activate
    
    
For n = 1 To gMax_Tie_Ref
        Cells(First + n - 1, gSumColSailNo).Value = Race_results(n, 1)     ' sail number
        Cells(First + n - 1, gSumColSailNo).Font.Color = QBColor(13)       ' sail number
        Cells(First + n - 1, gSumColSailNo).Font.Bold = True               ' sail number
        Cells(First + n - 1, gSumColSailNo).HorizontalAlignment = xlCenter ' sail number
        
        
For m = 1 To BestScores               'Best scores excluding discard & Race 1 (depends on gHMS)
            Cells(First + n - 1, gSumColSailNo + m).Value = Race_results(n, m + 1)
            Cells(First + n - 1, gSumColSailNo + m).Font.Color = QBColor(13)
            Cells(First + n - 1, gSumColSailNo + m).Font.Bold = True
        
Next m
        
        
For m = BestScores + 1 To BestScores + gLast_Race             'Race scores in reverse order
            Cells(First + n - 1, gSumColSailNo + m).Value = Race_results(n, m + 1)
            Cells(First + n - 1, gSumColSailNo + m).Font.Color = QBColor(9)
            Cells(First + n - 1, gSumColSailNo + m).Font.Bold = True
        
Next m
        Cells(First + n - 1, gSumColTieRef).Font.Color = QBColor(0)       ' Tie ref column
    
Next n



End Sub

Sub Break_Ties(position As Integer, FirstR As Integer, LastR As Integer, FirstC As Integer, LastC As Integer, SortC As Integer)

    
Dim m As Integer, n As Integer
    
Dim RowsLeft As Integer
    
Dim NumSame As Integer, SetRows As Integer, RowCount As Integer
    
Dim SetFirstR As Integer, SetLastR As Integer, colB As Integer, colF As Integer
    
    
Dim rRange As Range
        
    
If Not TypeName(ActiveSheet) = "Summary" Then Sheets("Summary").Activate
    
    '
Set size of basic array to have ties broken -the Group of Ties
    RowsLeft = LastR - FirstR + 1           'Rows left to be dealt with in Tied Group
     
    SetFirstR = FirstR
    SetLastR = LastR
    colB = 2                                ' column B
    colF = 6                                ' column F
    
    
Do          'To deal with a 
Set having same best score in Group of Ties
        Array_sort SetFirstR, FirstC, SetLastR, LastC, SortC
        NumSame = SameAs(SetFirstR, SortC, SetLastR)
        
If NumSame = 1 Then
            Cells(SetFirstR, colB).Value = position          'label as best
            Cells(SetFirstR, colF).Value = position          'label as best
            position = position + 1
            Cells(SetFirstR, SortC).Font.Color = QBColor(12) ' red
            Cells(SetFirstR, SortC).Font.Bold = True         ' bold
            Cells(SetFirstR, SortC).Interior.colorIndex = 8  ' cyan background
            RowsLeft = RowsLeft - 1          'One less to deal with in Group
            SetFirstR = SetFirstR + 1
            SetLastR = LastR
            SortC = gSumColBestL             ' column for Best Score
        Else
            SetLastR = SetFirstR + NumSame - 1
            SortC = SortC + 1
            
If SortC > gSumColRevR - gHMS Then                ' exclude Race 1 if HMS mode
                
For n = 0 To NumSame - 1
                    
Set rRange = Range(Cells(SetFirstR + n, gSumColTieNo), Cells(SetFirstR + n, gSumColSailNo))
                    rRange.ClearContents
                    rRange.Merge
                    rRange.Value = "Can't break tie No. " & gTie_Number & "; Toss for it!"
                    rRange.Font.Color = QBColor(gTColour)
                    gStill_tied = True                        'added 21/10/04 HCF
                
Next n
                RowsLeft = RowsLeft - NumSame
                SetFirstR = SetFirstR + NumSame
                SetLastR = LastR
                SortC = gSumColBestL                          ' reset to column 10 = Best Score
                position = position + NumSame
                'Swap between Red & Blue for different tied Sets
                
If gTColour = 12 Then gTColour = 9 
Else gTColour = 12
            
End If
        
End If
    
Loop Until RowsLeft = 0     'End of Group

End Sub


Function SameAs(startrow As Integer, StartCol As Integer, endrow As Integer) As Integer
    'find number of rows with same best score in chosen column
    
Dim n As Integer
    SameAs = 1
    
For n = 1 To endrow - startrow
        
If Cells(startrow, StartCol).Value = (Cells(startrow + n, StartCol).Value) Then SameAs = SameAs + 1
    
Next n

End Function

Sub Sort_Entrants(C1 As String)
    
    
Dim sailNumberRow As Integer, iAnswer As Integer, LastRow As Integer, x As Integer
    
    
If Range("A9").Value < 2 Then
        iAnswer = MsgBox("You have too few skippers - must have at least 2", vbOKOnly)
        Exit Sub
    
End If
        
    Call validateDataEntry
    
If gfoundError Then Exit Sub
        
    ActiveSheet.Unprotect (gPassword)
    
    
If C1 = "A10" Then
        Range("A10").FormulaR1C1 = "=RC[57]"         ' A10: =BF10 position
    
ElseIf C1 = "B10" Then
        Range("A10").FormulaR1C1 = "=RC[+1]"         ' A10: =B10 names
    
ElseIf C1 = "C10" Then
        Range("A10").FormulaR1C1 = "=VALUE(RC[+2])"  ' A10: =C10 sail num
    
ElseIf C1 = "D10" Then
        Range("A10").FormulaR1C1 = "=RC[+3]"         ' A10: =D10 club
    
ElseIf C1 = "E10" Then
        Range("A10").FormulaR1C1 = "=RC[+4]"         ' A10: =E10 hull
    
ElseIf C1 = "F10" Then
        Range("A10").FormulaR1C1 = "=VALUE(RC[+5])"  ' A10: =F10 freq
    
ElseIf C1 = "G10" Then
        Range("A10").FormulaR1C1 = "=VALUE(RC[+7])"  ' A10: =H10 scores
    
End If
    
    'copy formula down column A
    Range("A10").AutoFill Destination:=Range("A10:A93"), Type:=xlFillDefault
    
        ' in case boats were removed after initial entry, we may have blank lines that mess up
        ' the count of boats kept in A9 - we'll sort those out first - problem identified and
        ' solution provided by Henry Farley - 6aug2004
        ' also for sail numbers we want to perform an alpha sort on C10 before we perform
        ' the numeric sort on A10
        Range(Cells(10, 1), Cells(93, 7)).Sort Key1:=Range("C10"), Order1:=xlAscending, Header:=xlNo, _
            OrderCustom:=1, MatchCase:=False, Orientation:=xlTopToBottom
    
    Range(Cells(10, 1), Cells(9 + Cells(9, 1).Value, 7)).Sort Key1:=Range("A10"), Order1:=xlAscending, Header:=xlNo, _
    OrderCustom:=1, MatchCase:=False, Orientation:=xlTopToBottom
    
    ' make sure that all values in Sail column are strings (Var
Type = 8) otherwise Data
    ' Validation may appear to not work correctly
    ' this happens when user performed a cut and paste special to bring in values
    ' from another worksheet (e.g., previous regatta)
    LastRow = Cells(9, 1).Value + 9
    
For sailNumberRow = 10 To LastRow
        
If Not VarType(Range("C" & sailNumberRow).Value) = 8 Then
            Range("C" & sailNumberRow).Value = CStr(Range("C" & sailNumberRow).Value)
        
End If
    
Next sailNumberRow
   
    
    ' reset column A to rank value located in column BF
    Range("A10").FormulaR1C1 = "=RC[57]"              ' A10: =BE10
    Range("A10").AutoFill Destination:=Range("A10:A93"), Type:=xlFillDefault
    ActiveWindow.ScrollRow = 10
    
    Range("B10:C93").HorizontalAlignment = xlCenter
    Range("B10:B93").HorizontalAlignment = xlLeft
    Columns("A:H").EntireColumn.AutoFit
    Range("A9").Select
    Call Color_Discards
    ActiveSheet.Unprotect (gPassword)
    
If gStill_tied = False Then
        
If Range("G9").Value < 1 Then   ' manual tie break not used to break ties
            Columns("G").Hidden = True  ' hide manual tie break column
        
Else                            ' manual tie break used to break ties so
            Columns("G").Hidden = False ' display manual tie break column
        
End If
    
Else                                ' we still have tied results so
        Columns("G").Hidden = False     ' display manual tie break column
    
End If
    ActiveSheet.Protect (gPassword)

End Sub

Sub VerifyRace_Entry(C1 As String, RaceNum As Integer)
    ' Routine added 6 Feb 2004 by Herman van Beek
    
Dim x As Integer
    
Dim y As Integer
    
Dim C1Row As Integer
    
Dim ARow As Integer
    
Dim C1End As Integer
    
Dim AEnd As Integer
    
Dim Promote As Integer
    
Dim promoCol As Integer
    
Dim RaceScoreCol As Integer
    
Dim lastRowScoreSheet As Integer
    
    Worksheets("Race results").Activate
    gRaceNum = RaceNum
    gValidResult = True
    promoCol = 5 + (RaceNum - 1) * 4                        ' third column
   
    
    
If RaceNum = 1 Then
        Promote = 0
        
    Else
        Promote = Cells(5, promoCol).Value                  ' row 5 has promotion for this race
        
If Promote < 2 Then Promote = gPromote              ' 4 = default
        
If Promote > 18 Then Promote = gPromote             ' 4 = default
    
End If
    
    lastRowScoreSheet = 9 + gNo_Boats                       ' last row number on Score Sheet
    AEnd = 1 + gNo_Boats
    
If AEnd = 1 Then
        x = MsgBox("No Boats in the 'Valid List'", vbOKOnly, "Validate results")
    
End If

    Worksheets("Race results").Activate
    ActiveSheet.Unprotect (gPassword)
    ' clear out any previous entries in our scratch memory - column FL on ''Race results''
    Range("'Race results'!FL1:FL93").ClearContents
        
    ' check if Heat A has entries
    
If Range(C1 & "7").Value = 0 Then
        x = MsgBox("No Sail Numbers available for Race " & gRaceNum, vbOKOnly, "Did you enter results?")
    Else
        Call VerifyHeat_Entry(C1, "A", 8, 31, RaceNum)                                 ' Heat A
        
If Range(C1 & "32").Value <> "" Then
            Call VerifyHeat_Entry(C1, "B", 33 + Promote, 56, RaceNum)                  ' Heat B
            
If Range(C1 & "57").Value <> "" Then
                Call VerifyHeat_Entry(C1, "C", 58 + Promote, 81, RaceNum)              ' Heat C
                
If Range(C1 & "82").Value <> "" Then
                    Call VerifyHeat_Entry(C1, "D", 83 + Promote, 106, RaceNum)          ' Heat D
                    
If Range(C1 & "107").Value <> "" Then
                        Call VerifyHeat_Entry(C1, "E", 108 + Promote, 131, RaceNum)     ' Heat E
                        
If Range(C1 & "132").Value <> "" Then
                            Call VerifyHeat_Entry(C1, "F", 133, 156, RaceNum) ' Heat F
                        
End If
                    
End If
                
End If
            
End If
        
End If
        
'--------------------------------------------------------------------------
        ' Make sure every boat appears exactly once
        
For ARow = 2 To AEnd
            
If Range("'Race results'!FL" & ARow).Value < 1 Then
                gValidResult = False
                x = MsgBox("No entry found for boat: " & Range("A" & ARow).Value & " in Race " & gRaceNum, vbOKOnly, "Did you make a typo?")
            
ElseIf Range("'Race results'!FL" & ARow).Value > 1 Then
                gValidResult = False
                x = MsgBox("Boat " & Range("A" & ARow).Value & " scored " & Range("'Race results'!FL" & ARow).Value & " times", vbOKOnly, "Ooops! Scored more than once in Race " & gRaceNum)
            
End If
        
Next ARow
        
If gValidResult = True Then
            ' Now check that column does not contain "" indicating a Scoring Error
            ' a null field is most likely caused by a VLOOKUP error
            ' Added 17 July 2007 by HCF and updated by HvB 22 July 2008
        
            RaceScoreCol = 8 + RaceNum
            
For ARow = 10 To lastRowScoreSheet
                
If Worksheets("Score Sheet").Range(ColumnLetter(RaceScoreCol) & ARow).Value = "" Then
                    x = MsgBox("The results for Race " & gRaceNum & " show one or more PROBLEMS " _
                    & Chr$(13) & " - all boats accounted for;" _
                    & Chr$(13) & " - no duplicate entries;" _
                    & Chr$(13) & " - PROBLEM: Boat " & Range("'Score Sheet'!C" & ARow).Value & " in Race " & RaceNum & " did not race after being PROMOTED " _
                    & Chr$(13) & " " _
                    & Chr$(13) & " It might be that results for Heat D entered in Heat C or something similar", _
                    vbOKOnly, "SCORING PROBLEM")
                
                    ActiveSheet.Protect (gPassword)
                    Exit Sub
                
End If
            
Next ARow
        
            x = MsgBox("The results for Race " & gRaceNum & " are OK " _
                & Chr$(13) & " - all boats accounted for;" _
                & Chr$(13) & " - no duplicate entries;" _
                & Chr$(13) & " - all boats sailed after being promoted from previous heat;", _
                vbOKOnly, "Congratulations!!!")
            Range(C1 & "8").Select
        
End If
        
'--------------------------------------------------------------------------
    
End If
    ActiveSheet.Protect (gPassword)

End Sub

Sub VerifyHeat_Entry(C1 As String, Heat As String, startrow As Integer, endrow As Integer, RaceNum As Integer)
    ' Routine added 6 Feb 2004 by Herman van Beek
    
Dim x As Integer, C1Row As Integer
    
Dim ARow As Integer
    
Dim AEnd As Integer
    
Dim ValidBoat As Boolean
    
    AEnd = 1 + Range("'Score Sheet'!A9").Value        ' row containing last boat
    ' Go through the list of boats in column C1
    
For C1Row = startrow To endrow                    ' skips over first four slots in heats B - F
        
If Range(C1 & C1Row).Value <> "" Then
            ValidBoat = False
            ' Verify the boat is in column "A", the list of all boats
            
For ARow = 2 To AEnd
                
If Range(C1 & C1Row).Value = Range("A" & ARow).Value Then
                    ' The boat was found, stop search
                    ValidBoat = True
                    ' keep track in column FL of how many times this boat was found
                    Range("'Race results'!FL" & ARow).Value = Range("'Race results'!FL" & ARow).Value + 1
                    Exit For
                
End If
            
Next ARow
            
If ValidBoat = False Then
                ' The boat was not found, report error
                gValidResult = False
                x = MsgBox("Race " & gRaceNum & ": Invalid Boat Entered in Heat " & Heat & ": " & Range(C1 & C1Row).Value, vbOKOnly, "Data Entry Verification")
                Range(C1 & C1Row).Select
                Exit For
            
End If
        
End If
    
Next C1Row
    ' future enhancement...
    ' make sure that if we have multiple letter scores that they are in the proper order
    ' as defined by HMS2006.  This order is shown in row 150/151 on Race Results sheet
    ' 
For C1Row = StartRow To EndRow
    ' 
Next C1Row

End Sub

Sub Reset_Race_Results(Race As Integer)
    
Dim Promote As Integer, promoCol As Integer, pointsCol As Integer, expCol As Integer
    
        
    'NOTE:  Logic depends on finding either constants in the Comments and Points fields
    '       or correct equations - it takes the first equation found by looking down those
    '       columns and then copying it to all fields in that column.
    
    ' Each race is 4 columns wide.  The header field of 4th column in each heat contains number
    ' of scores entered for that heat.  Row 117 starts WDN "heat" so if 4th column of any race in row 117
    ' shows "01" then we have no entries for that race and, hence, don't need to reset that race unless we're
    ' changing promotion

    
If Not TypeName(ActiveSheet) = "Race results" Then Sheets("Race results").Activate
    ' Race #1 starts in column "C" (3)
    gCol = 3 + (Race - 1) * 4               ' calculate starting column for race
    expCol = gCol + 3                       ' fourth column contains points
        
    ' we found race entries so time to go to work...
    
If Sheets("Race results").ProtectContents Then Sheets("Race results").Unprotect (gPassword)
    Application.DisplayStatusBar = True
    gColLetter = ColumnLetter(gCol)         ' calculate starting column letter for race
    promoCol = gCol + 2                     ' third column contains promotion value
    gColLetterp2 = ColumnLetter(promoCol)   ' calculate column letter for points and promotion value
    gColLetterp3 = ColumnLetter(expCol)     ' calculate column letter for exp
    gRaceNum = Race                         ' set global variable for possible use in error message
    gChangeRaceResults = True               ' set flag we probably changed something
   
   
    
If Race = 1 Then
        Promote = 0
       
    Else
        Promote = Cells(5, promoCol).Value                  ' row 5 has promotion
        
If Promote < 2 Then Promote = 4                     ' 4 = default
        
If Promote > 18 Then Promote = 4                    ' 4 = default
    
End If
    Application.StatusBar = "Resetting rows for actual finishes in Race " & Race & " Heat A"
    Call Reset_Heat_Results(8, 31, "Up")                  ' Heat A
    Application.StatusBar = "Resetting rows for actual finishes in Race " & Race & " Heat B"
    Call Reset_Heat_Results(33 + Promote, 56, "Up")       ' Heat B
    Application.StatusBar = "Resetting rows for actual finishes in Race " & Race & " Heat C"
    Call Reset_Heat_Results(58 + Promote, 81, "Up")       ' Heat C
    Application.StatusBar = "Resetting rows for actual finishes in Race " & Race & " Heat D"
    Call Reset_Heat_Results(83 + Promote, 106, "Up")       ' Heat D
    Application.StatusBar = "Resetting rows for actual finishes in Race " & Race & " Heat E"
    Call Reset_Heat_Results(108 + Promote, 131, "Up")      ' Heat E
    Application.StatusBar = "Resetting rows for withdrawn boats in Race " & Race
    Call Reset_Heat_Results(133, 156, "Up")      ' Withdrawn Boats
    
    
If Race > 1 Then                             ' take care of UP/green promotion region
        Application.StatusBar = "Resetting rows for promoted boats in Race " & Race & " Heat B"
        Call Reset_Heat_Results(33, 32 + Promote, "Down")   ' Heat B
        Cells(32, gCol).Formula = "=COUNTA(" & gColLetter & (33 + Promote) & ":" & gColLetter & "56)"
        Cells(32, gCol + 3).Formula = "=COUNTA(" & gColLetter & (33 + Promote) & ":" & gColLetter & "56)-" & gColLetterp2 & "32+" & gColLetterp3 & "7"

Sub Reset_Heat_Results(RowStart As Integer, RowEnd As Integer, Direction As String)
    
Dim x As Integer, iRow As Integer
    
Dim bFoundEq As Boolean
    
Dim rRange As Range
    
    ' We're on the "Race results" sheet
    
Set rRange = Range(Cells(RowStart, gCol), Cells(RowEnd, gCol))
    rRange.ClearContents  ' clear out sail numbers
    
    
With rRange.Validation
        .Delete
        .Add Type:=xlValidateList, AlertStyle:=xlValidAlertStop, Operator:= _
        xlBetween, Formula1:="=Sail_No.s"
        .IgnoreBlank = False
        .InCellDropdown = True
        .InputTitle = ""
        .ErrorTitle = "NO SUCH BOAT!"
        .InputMessage = ""
        .ErrorMessage = "There is no boat with that Sail Number in the event."
        .ShowInput = True
        .ShowError = True
    
End With
    
    ' Rebuild equation in the Comments column -
    ' if Direction = Down - use formula in RowStart and duplicate down to RowEnd
    ' if Direction = Up   - use formula in RowEnd and duplicate up to RowStart
    bFoundEq = False
    
If Direction = "Down" Then
        
If Cells(RowStart, gCol + 1).HasFormula Then
            bFoundEq = True
            Cells(RowStart, gCol).AutoFill Destination:=Range(Cells(RowStart, gCol), Cells(RowEnd, gCol)), Type:=xlFillFormats
            Cells(RowStart, gCol + 1).AutoFill Destination:=Range(Cells(RowStart, gCol + 1), Cells(RowEnd, gCol + 1)), Type:=xlFillSeries
        
End If
    Else
        
If Cells(RowEnd, gCol + 1).HasFormula Then
            bFoundEq = True
            Cells(RowEnd, gCol).AutoFill Destination:=Range(Cells(RowStart, gCol), Cells(RowEnd, gCol)), Type:=xlFillFormats
            Cells(RowEnd, gCol + 1).AutoFill Destination:=Range(Cells(RowStart, gCol + 1), Cells(RowEnd, gCol + 1)), Type:=xlFillSeries
        
End If
    
End If
    ' if we did not find any equations, we need manual intervention
    
If Not bFoundEq Then
        x = MsgBox("Manual intervention required - did not find equation in Comment column for Race " & gRaceNum & " Row " & RowStart, vbOKOnly, "Problems!!!")
    
End If
    
    ' Now do the same for the Points column
    bFoundEq = False
    
If Direction = "Down" Then
        
If Cells(RowStart, gCol + 2).HasFormula Then
            bFoundEq = True
            Cells(RowStart, gCol + 2).AutoFill Destination:=Range(Cells(RowStart, gCol + 2), Cells(RowEnd, gCol + 2)), Type:=xlFillSeries
            Cells(RowStart, gCol + 3).AutoFill Destination:=Range(Cells(RowStart, gCol + 3), Cells(RowEnd, gCol + 3)), Type:=xlFillFormats
        
End If
    Else
        
If Cells(RowEnd, gCol + 2).HasFormula Then
            bFoundEq = True
            Cells(RowEnd, gCol + 2).AutoFill Destination:=Range(Cells(RowEnd, gCol + 2), Cells(RowStart, gCol + 2)), Type:=xlFillSeries
            Cells(RowEnd, gCol + 3).AutoFill Destination:=Range(Cells(RowEnd, gCol + 3), Cells(RowStart, gCol + 3)), Type:=xlFillFormats
        
End If
    
End If
    ' if we did not find any equations, we need manual intervention
    
If Not bFoundEq Then
        x = MsgBox("Manual intervention required - did not find equation in Points column for Race " & gRaceNum & " Row " & RowStart, vbOKOnly, "Problems!!!")
    
End If

End Sub

Function ColumnLetter(ColumnNumber As Integer) As String
  
If ColumnNumber > 26 Then

    ' 1st character:  Subtract 1 to map the characters to 0-25,
    '                 but you don't have to remap back to 1-26
    '                 after the 'Int' operation since columns
    '                 1-26 have no prefix letter

    ' 2nd character:  Subtract 1 to map the characters to 0-25,
    '                 but then must remap back to 1-26 after
    '                 the 'Mod' operation by adding 1 back in
    '                 (included in the '65')

    ColumnLetter = Chr(Int((ColumnNumber - 1) / 26) + 64) & _
                   Chr(((ColumnNumber - 1) Mod 26) + 65)
  Else
    ' Columns A-Z
    ColumnLetter = Chr(ColumnNumber + 64)
  
End If

End Function

Sub Color_Discards()
    '
    ' Highlight the discard cells -- we're on the Score Sheet
    
    
Dim resultRow As Integer, minRow As Integer, maxRow As Integer
    
Dim mincalcCol As Integer, maxcalcCol As Integer, calcCol As Integer, i As Integer, colorIndex As Integer
    
Dim discard(1 To 6) As Variant
    
Dim DNEArtificial_1 As Variant, DNEArtificial_2 As Variant
    
Dim MAXPointsRace1 As Variant, MAXPointsRaceN As Variant, MAXPoints As Variant
    
Dim scoringProblem As Boolean
    
    gLast_Race = Range("'Score Sheet'!BS9").Value
    scoringProblem = False
    minRow = 10
    maxRow = 9 + Range("A9").Value
    mincalcCol = Range("I10:AW93").Column                   ' column I
    maxcalcCol = Range("I10:AW93").Column + gLast_Race - 1  ' column containing the last race
    DNEArtificial_1 = Range("'Race results'!L2").Value      ' score for DNE in Races 2 - N
    DNEArtificial_2 = Range("'Race results'!L3").Value      ' score for DNE in Race 1
    MAXPointsRaceN = Range("'Race results'!L2").Value       ' max possible score for Races 2 - N
    MAXPointsRace1 = Range("'Race results'!L3").Value       ' max possible score for Race 1
                
    ' clear the cells first
    Call clearBackgroundColor
    
    '           magenta (7)   green (10)
    '           yellow (6)    dark blue (11)
    '           red (3)       mustard (12)
    '           salmon (22)   purple (13)
    '           cyan (8)      teal (14)
    '           brown (9)     light gray (15)
        
    
For resultRow = minRow To maxRow
        discard(1) = Range("AY" & resultRow).Value      ' col AY
        discard(2) = Range("AZ" & resultRow).Value      ' col AZ
        discard(3) = Range("BA" & resultRow).Value      ' col BA
        discard(4) = Range("BB" & resultRow).Value      ' col BB
        discard(5) = Range("BC" & resultRow).Value      ' col BC
        discard(6) = Range("BD" & resultRow).Value      ' col BD
        
For i = 1 To 6
            
If discard(i) <> "" Then
                
For calcCol = mincalcCol To maxcalcCol
                    
If Cells(resultRow, calcCol).Value = discard(i) Then
                        
If Cells(resultRow, calcCol).Interior.colorIndex <> 15 Then
                            Cells(resultRow, calcCol).Interior.colorIndex = 15
                            Cells(resultRow, calcCol).Font.colorIndex = xlAutomatic
                            Exit For
                        
End If
                    
End If
                
Next calcCol
            
End If
        
Next i
        
For calcCol = mincalcCol To maxcalcCol
            
If calcCol = mincalcCol Then
                MAXPoints = MAXPointsRace1
            Else
                MAXPoints = MAXPointsRaceN
            
End If
        ' color all cells with DNE or DGM scores red, including Race #1
            
If Cells(resultRow, calcCol).Value = DNEArtificial_1 Or _
               Cells(resultRow, calcCol).Value = DNEArtificial_2 Then ' color DNE and DGM scores red
                
If Cells(resultRow, calcCol).Interior.colorIndex <> 3 Then
                    Cells(resultRow, calcCol).Interior.colorIndex = 3
                    Cells(resultRow, calcCol).Font.Color = QBColor(14) 'light yellow
                
End If
            
End If
            
If IsNumeric(Cells(resultRow, calcCol).Value) = False Then
                ' color background of empty cells salmon
                
If Cells(resultRow, calcCol).Value = "" Then
                    
If Cells(resultRow, calcCol).Interior.colorIndex <> 22 Then
                        Cells(resultRow, calcCol).Interior.colorIndex = 22
                        scoringProblem = True
                    
End If
                Else
                    ' color Awaiting Redress columns Magenta with yellow text
                    
If Cells(resultRow, calcCol).Interior.colorIndex <> 7 Then
                        Cells(resultRow, calcCol).Interior.colorIndex = 7
                        Cells(resultRow, calcCol).Font.Color = QBColor(14)
                        scoringProblem = True
                    
End If
                
End If
            Else
                ' flag bad values - color cells salmon
                
If Cells(resultRow, calcCol).Value < 1 Or Cells(resultRow, calcCol).Value > MAXPoints Then
                    
If Cells(resultRow, calcCol).Interior.colorIndex <> 22 Then
                        Cells(resultRow, calcCol).Interior.colorIndex = 22
                        scoringProblem = True
                    
End If
                
End If
            
End If
        
Next calcCol
    
Next resultRow
    
If scoringProblem Then
        Range("'Score Sheet'!H7").Interior.colorIndex = 3
        Range("'Score Sheet'!H7").Font.Color = QBColor(14)
        Range("'Score Sheet'!H8").Interior.colorIndex = 3
        Range("'Score Sheet'!H8").Font.Color = QBColor(14)
        Range("'Score Sheet'!H7").Value = "Scoring"
        Range("'Score Sheet'!H8").Value = "Problem"
    Else
        Range("'Score Sheet'!H7").Value = ""
        Range("'Score Sheet'!H7").Interior.colorIndex = 40
        Range("'Score Sheet'!H8").Interior.colorIndex = 37
        Range("'Score Sheet'!H8").Font.Color = vbBlack
        Range("'Score Sheet'!H8").Value = "Score"
    
End If
    Sheets("Score Sheet").Protect (gPassword)
    Range("A1").Select

End Sub

Sub P2W()
    '
    ' this code handles Print2Web for Excel 2000+
    ' NOTE: worksheet "HTML Source" must exist before calling this code!
    
    
Dim eventName As String, eventDate As String, eventHost As String, x As Integer
    
    eventName = Range("'Score Sheet'!B2").Value
    eventDate = Range("'Score Sheet'!B3").Value
    eventHost = Range("'Score Sheet'!B4").Value

              8 + gLast_Race)).Address

        SourceType:=xlSourceRange, _
        Filename:=ActiveWorkbook.Path & "/" & eventName & ".htm", _
        Sheet:=ActiveSheet.Name, _
        Source:=ActiveSheet.PageSetup.PrintArea, _
        HtmlType:=xlHtmlStatic, _
        DivID:="", _
        Title:=eventName & Chr(10) & eventDate & Chr(10) & "Hosted by" & Chr(10) & eventHost)
        .Publish (True)
    
End With
    
    ' we're done with this sheet so delete it
     Application.DisplayAlerts = False
    Worksheets("HTML Source").Delete
    Application.DisplayAlerts = True
    
    x = MsgBox("HTML file: " & eventName & ".htm created in " & ActiveWorkbook.Path & " directory", vbOKOnly, "Finished creating Web output!!!")

    Range("H10").Select

Sub SortByBoat_Entry()
    
Dim x As Integer
    ' collect sailnumbers from column C on Score Sheet and write out to column A on Race results sheet
    ' sort range and adjust range of "Sail_No.s" so list in sail number validation excludes extra empty spaces
    On Error GoTo errorhandler
    ' only perform this sort if there has been a change in value on the Score Sheet in column C (3)
    
If gChangeSailNumber = False Then Exit Sub
    
    gNo_Boats = Range("'Score Sheet'!A9").Value
    Application.EnableEvents = False
    
If Not TypeName(ActiveSheet) = "Race results" Then Sheets("Race results").Activate
    
If Sheets("Race results").ProtectContents Then Sheets("Race results").Unprotect (gPassword)
    Range("Sail_No.s").ClearContents            ' clear out any old data
    Range("SailSort").ClearContents             ' clear out any old data
    ' copy sail numbers from Score Sheet and write on Race results sheet
    '
    ' Range FN2:FN85 has been formatted "Number"
    ' Range FO2:FO85 has been formatted "Text"
    ' We'll sort on first column and then copy second column back to Column A
    ' this should properly handle sail numbers with leading zeros
    
    
For x = 1 To gNo_Boats
        Range("'Race results'!FN" & (1 + x)).Value = Range("'Score Sheet'!C" & (9 + x)).Value
        Range("'Race results'!FO" & (1 + x)).Value = Range("'Score Sheet'!C" & (9 + x)).Value
    
Next x
    
    ' Sort sail numbers - we'll sort on first column
    
If gNo_Boats > 1 Then
        Range("SailSort").Sort Key1:=Range("FN2"), Order1:=xlAscending, _
                Header:=xlNo, OrderCustom:=1, MatchCase:=False, Orientation:=xlTopToBottom
    
End If
    ' copy sorted sail numbers to Column A
    
For x = 1 To gNo_Boats
        Range("'Race results'!A" & (1 + x)).Value = Range("'Race results'!FO" & (1 + x)).Value
    
Next x
       
    ' Dynamically adjust "Sail_No.s" range
    
With Range("Sail_No.s")
        .Range("A1", .Range("A85").End(xlUp)).Name = "Sail_No.s"  ' adjust range limit to first blank field
    
End With
    
    gChangeSailNumber = False                          ' reset flag
    Sheets("Race results").Protect (gPassword)
    Application.StatusBar = False
    Application.EnableEvents = True
    Exit 
Sub      ' Skip over the errorhandler routine.

errorhandler:                   ' End the procedure if no comment in the cell.
    Exit Sub

End Sub
